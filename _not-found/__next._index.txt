// EU user consent policy wants a real answer, and granted everywhere else, which is what
// the privacy policy discloses. Google resolves the visitor's region itself from the
// request, so none of this needs an IP lookup of ours. Roughly two thirds of paying
// customers bill from the denied set, so it is the majority path, not an edge case.
//
// It governs GOOGLE tags only. The Meta Pixel and Microsoft Clarity keep the behaviour
// they have always had, which is a separate decision rather than an oversight.
//     https://developers.google.com/tag-platform/security/guides/consent

    // is ours and cannot be redefined out from under the banner.
    window.strideGtag = gtag;

    // the UK and Switzerland. Kept in step with _CONSENT_REQUIRED_COUNTRIES in

    var CONSENT_REGION = ['AT','BE','BG','HR','CY','CZ','DK','EE','FI','FR','DE','GR',

    function state(value, extra) {
        var o = {};
        for (var i = 0; i < ALL.length; i++) { o[ALL[i]] = value; }
        for (var k in (extra || {})) { o[k] = extra[k]; }
        return o;
    }

    // Region-scoped default first; Google applies the more specific one.
    // wait_for_update holds the tags briefly so a stored answer below lands first.
    gtag('consent', 'default', state('denied', { region: CONSENT_REGION,
                                                 wait_for_update: 500 }));
