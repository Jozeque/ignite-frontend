// Consent mode v2. This must run BEFORE any Google tag: after one has loaded it has
// already decided how to behave, and a declaration made later is decoration.
//
// Two defaults are declared. Denied for the EEA, the UK and Switzerland, where Google's
// EU user consent policy wants a real answer, and granted everywhere else, which is what
// the privacy policy discloses. Google resolves the visitor's region itself from the
// request, so none of this needs an IP lookup of ours. Roughly two thirds of paying
// customers bill from the denied set, so it is the majority path, not an edge case.
//
// It governs GOOGLE tags only. The Meta Pixel and Microsoft Clarity keep the behaviour
// they have always had, which is a separate decision rather than an oversight.
//     https://developers.google.com/tag-platform/security/guides/consent
(function () {
    window.dataLayer = window.dataLayer || [];
    function gtag() { dataLayer.push(arguments); }
    // A stable handle. The GA4 block further down declares its own global gtag; this one
    // is ours and cannot be redefined out from under the banner.
    window.strideGtag = gtag;

    // Where Google's EU user consent policy applies: the EU 27, the rest of the EEA,
    // the UK and Switzerland. Kept in step with _CONSENT_REQUIRED_COUNTRIES in
    // firebase_cloud/functions/google_dm.py, which decides the same question server-side.
    var CONSENT_REGION = ['AT','BE','BG','HR','CY','CZ','DK','EE','FI','FR','DE','GR',

        var o = {};
        for (var i = 0; i < ALL.length; i++) { o[ALL[i]] = value; }
        for (var k in (extra || {})) { o[k] = extra[k]; }
        return o;

    // wait_for_update holds the tags briefly so a stored answer below lands first.

                                                 wait_for_update: 500 }));

        // on the way in, because the bar may have arrived during that beat, and show it

            ready = true;
            if (!answered && !blocked(clearBottomBar())) { show(); } else { place(); }
        }, 700);

  var STRIDE_LOCALES = ['de', 'es', 'fr', 'it', 'ja', 'zh'];

  function strideLangCurrent() {
    var seg = location.pathname.split('/')[1];

  function strideLangPath(code) {
    var p = location.pathname;
    var cur = strideLangCurrent();
    if (cur !== 'en') p = p.slice(cur.length + 1) || '/';
    if (code !== 'en') p = '/' + code + (p === '/' ? '/' : p);
    return p + location.search + location.hash;
